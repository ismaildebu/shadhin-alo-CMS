    /** একটি খবরের কার্ড তৈরির পেজ। */
    public function show(News $news): View
    {
        Gate::authorize('view', $news);

        $news->loadMissing('category');

        return view('admin.news.card', [
            'news' => $news,
            'card' => $this->payload($news),
            'defaults' => $this->currentDefaults(),
            'templates' => self::TEMPLATES,
        ]);
    }

    /** ব্রাউজারে আঁকা কার্ড (JPG) সংরক্ষণ। */
    public function store(Request $request, News $news): JsonResponse|RedirectResponse
    {
        Gate::authorize('update', $news);

        $request->validate([
            'card' => ['required', 'file', 'mimes:jpg,jpeg,png', 'max:6144'],
        ], [
            'card.required' => 'কার্ডের ছবি পাওয়া যায়নি।',
            'card.mimes' => 'কার্ড jpg বা png হতে হবে।',
            'card.max' => 'কার্ডের ছবি সর্বোচ্চ ৬ MB হতে পারবে।',
            'card.uploaded' => 'আপলোড ব্যর্থ হয়েছে (সার্ভারের আপলোড সীমা কম হতে পারে)।',
        ]);

        $file = $request->file('card');
        $name = $news->id.'-'.Str::uuid().'.'.$file->extension();
        $file->move(public_path(self::DIR), $name);

        $old = $news->card_image;

        // সরাসরি কোয়েরি: খবরের updated_at বা অন্য ইভেন্ট ছোঁয়া হয় না।
        News::whereKey($news->id)->update(['card_image' => $name]);

        $this->deleteFile($old);

        if ($request->expectsJson()) {
            return response()->json(['ok' => true, 'url' => '/'.self::DIR.'/'.$name]);
        }

        return back()->with('success', 'ফটোকার্ড সংরক্ষণ হয়েছে।');
    }

    public function destroy(Request $request, News $news): JsonResponse|RedirectResponse
    {
        Gate::authorize('update', $news);

        $old = $news->card_image;

        News::whereKey($news->id)->update(['card_image' => null]);

        $this->deleteFile($old);

        if ($request->expectsJson()) {
            return response()->json(['ok' => true]);
        }

        return back()->with('success', 'ফটোকার্ড মোছা হয়েছে।');
    }

    /** ফটোকার্ড কেন্দ্র (অ্যাডমিন ও এডিটর): ডিফল্ট ডিজাইন ও একসাথে তৈরি। */
    public function center(): View
    {
        return view('admin.photocards', [
            'total' => News::published()->count(),
            'withCard' => News::published()->whereNotNull('card_image')->count(),
            'recent' => News::published()->with('category')->latestPublished()->take(20)->get(),
            'defaults' => $this->currentDefaults(),
            'templates' => self::TEMPLATES,
            'queueUrl' => route('photocards.queue', [], false),
            'footerFallback' => $this->footerText(),
        ]);
    }

    /** কার্ড নেই এমন প্রকাশিত খবরের তালিকা (JSON), কেন্দ্রের "সব তৈরি করুন" বোতামের জন্য। */
    
    public function queue(): JsonResponse
    {
        $items = News::published()
            ->whereNull('card_image')
            ->with('category')
            ->latestPublished()
            ->take(100)
            ->get();

        return response()->json([
            'items' => $items->map(fn (News $n) => [
                'id' => $n->id,
                'title' => $n->title,
                'category' => $n->category->name ?? '',
                'color' => $n->category?->home_color,
                'date' => $n->display_date->format('d M Y'),
                'image' => $n->image ? '/uploads/news/'.$n->image : null,
                'storeUrl' => route('news.card.store', $n, false),
                'excerpt' => $this->plain($n->summary ?: $n->description, 280),
                'caption' => $this->caption($n),
                'byline' => (string) $n->reporter,
            ])->values(),
            'logo' => $this->path(Settings::logoUrl()),
            'emblem' => $this->emblem(),
            'siteName' => Settings::get('site_name'),
            'tagline' => Settings::get('site_tagline'),
            'footer' => $this->footerText(),
            'social' => $this->social(),
        ]);
    }

    public function saveDefaults(Request $request): RedirectResponse
    {
        $request->validate([
            
            'card_template' => ['required', Rule::in(array_keys(self::TEMPLATES))],
            'card_size' => ['required', Rule::in(['landscape', 'square', 'portrait'])],
            'card_accent' => ['required', 'regex:/^#[0-9a-fA-F]{6}$/'],
            'card_footer' => ['nullable', 'string', 'max:80'],
        ], [
            'card_accent.regex' => 'রংটি সঠিক নয়।',
        ]);

        Settings::set([
            'card_template' => (string) $request->input('card_template'),
            'card_size' => (string) $request->input('card_size'),
            'card_accent' => (string) $request->input('card_accent'),
            'card_footer' => trim((string) $request->input('card_footer', '')),
            'card_logo' => $request->boolean('card_logo') ? '1' : '0',
            'card_category' => $request->boolean('card_category') ? '1' : '0',
            'card_date' => $request->boolean('card_date') ? '1' : '0',
        ]);

        return back()->with('success', 'ডিফল্ট ডিজাইন সংরক্ষণ হয়েছে।');
    }

    private function payload(News $news): array
    {
        return [
            'id' => $news->id,
            'title' => $news->title,
            'category' => $news->category->name ?? '',
            'color' => $news->category?->home_color,
            'date' => $news->display_date->format('d M Y'),
            'image' => $news->image ? '/uploads/news/'.$news->image : null,
            'logo' => $this->path(Settings::logoUrl()),
            'emblem' => $this->emblem(),
            'siteName' => Settings::get('site_name'),
            'tagline' => Settings::get('site_tagline'),
            'footer' => $this->footerText(),
            'excerpt' => $this->plain($news->summary ?: $news->description, 280),
            'caption' => $this->caption($news),
            'byline' => (string) $news->reporter,
            'social' => $this->social(),
            'card' => $news->card_image ? '/'.self::DIR.'/'.$news->card_image : null,
            'storeUrl' => route('news.card.store', $news, false),
            'deleteUrl' => route('news.card.destroy', $news, false),
        ];
    }

    private function currentDefaults(): array
    {
        return [
            'template' => Settings::get('card_template'),
            'size' => Settings::get('card_size'),
            'logo' => Settings::get('card_logo') === '1',
            'category' => Settings::get('card_category') === '1',
            'date' => Settings::get('card_date') === '1',
            'accent' => Settings::get('card_accent'),
            'footer' => Settings::get('card_footer'),
        ];
    }

    /** কার্ডের নিচে ডানে যে লেখা বসবে: সেটিংয়ের লেখা, না থাকলে সাইটের ডোমেইন। */
    private function footerText(): string
    {
        $custom = Settings::get('card_footer');

        if ($custom !== '') {
            return $custom;
        }

        return (string) (parse_url((string) config('app.url'), PHP_URL_HOST) ?? '');
    }

    /** পূর্ণ ঠিকানা থেকে শুধু পথ (একই উৎসের ছবি, ক্যানভাস নিরাপদ থাকে)। */
    private function path(string $url): string
    {
        return parse_url($url, PHP_URL_PATH) ?: $url;
    }

    /** HTML থেকে সাধারণ লেখা: ট্যাগ বাদ, সংক্ষিপ্ত করা। */
    private function plain(?string $html, int $limit): string
    {
        $text = html_entity_decode(strip_tags((string) $html), ENT_QUOTES | ENT_HTML5, 'UTF-8');

        return Str::limit(trim((string) preg_replace('/\s+/u', ' ', $text)), $limit, '…');
    }

    private function caption(News $news): string
    {
        $caption = trim((string) $news->image_caption);
        $credit = trim((string) $news->image_credit);

        return trim($caption.($credit !== '' ? ' (ছবি: '.$credit.')' : ''));
    }

    /** গোল প্রতীকের জন্য ফেভিকন (সেট করা থাকলে)। */
    private function emblem(): ?string
    {
        $url = Settings::faviconUrl();

        return $url ? $this->path($url) : null;
    }

    /**
     * সাইট সেটিংয়ের সোশ্যাল লিংক থেকে হ্যান্ডেলের নাম (যেমন facebook.com/abc হলে "abc")।
     *
     * @return list<array{type: string, text: string}>
     */
    private function social(): array
    {
        $items = [];

        foreach (['facebook' => 'social_facebook', 'instagram' => 'social_instagram', 'youtube' => 'social_youtube', 'x' => 'social_x'] as $type => $key) {
            $url = Settings::get($key);

            if ($url === '') {
                continue;
            }

            $path = trim((string) parse_url($url, PHP_URL_PATH), '/');
            $handle = ltrim(Str::afterLast($path, '/'), '@');

            $items[] = [
                'type' => $type,
                'text' => $handle !== '' ? $handle : (string) parse_url($url, PHP_URL_HOST),
            ];
        }

        return $items;
    }

    private function deleteFile(?string $name): void
    {
        if ($name) {
            File::delete(public_path(self::DIR.'/'.basename($name)));
        }
    }
}

     * Normalized, safe configuration for the article page.
     *
     * @return array{
     *     opt: array<string, string>,
     *     main_col: string,
     *     side_col: ?string,
     *     toggles: array<string, bool>,
     *     sidebar: list<string>,
     *     counts: array<string, int>,
     *     css: string
     * }
     */
    public static function all(): array
    {
        $s = Settings::all();

        $opt = [
            'layout' => self::pick($s['article_layout'] ?? '', self::LAYOUTS, 'classic'),
            'side' => self::pick($s['article_sidebar_side'] ?? '', self::SIDES, 'right'),
            'font' => self::pick($s['article_font'] ?? '', self::FONTS, 'serif'),
            'title_size' => self::pick($s['article_title_size'] ?? '', self::TITLE_SIZES, 'md'),
            'body_size' => self::pick($s['article_body_size'] ?? '', self::BODY_SIZES, 'md'),
            'body_width' => self::pick($s['article_body_width'] ?? '', self::BODY_WIDTHS, 'normal'),
            'leading' => self::pick($s['article_leading'] ?? '', self::LEADINGS, 'normal'),
            'image_ratio' => self::pick($s['article_image_ratio'] ?? '', self::IMAGE_RATIOS, '16x9'),
            'image_fit' => self::pick($s['article_image_fit'] ?? '', self::IMAGE_FITS, 'cover'),
            'accent' => Colors::isHex($s['article_accent'] ?? null) ? $s['article_accent'] : self::DEFAULTS['article_accent'],
            'link' => Colors::isHex($s['article_link'] ?? null) ? $s['article_link'] : self::DEFAULTS['article_link'],
        ];

        $toggles = [];

        foreach (array_keys(self::TOGGLES) as $key) {
            $toggles[$key] = ($s['article_show_'.$key] ?? '1') === '1';
        }

        $hidden = self::csv($s['article_sidebar_hidden'] ?? '');
        // Only widgets present in the saved order are shown, so widgets added later stay hidden
        // until an editor saves the form.
        $sidebar = array_values(array_diff(self::csv($s['article_sidebar_order'] ?? ''), $hidden));

        // No sidebar parts selected: use the single-column grid so the page has no empty gap.
        $columns = self::LAYOUTS[$sidebar === [] ? 'single' : $opt['layout']];

        $counts = [];

        foreach (self::COUNTS as $key => $rule) {
            $counts[$key] = self::clamp($s['article_'.$key.'_count'] ?? '', $rule);
        }

        return [
            'opt' => $opt,
            'main_col' => $columns['main'],
            'side_col' => $columns['side'],
            'toggles' => $toggles,
            'sidebar' => $sidebar,
            'counts' => $counts,
            'css' => self::css($opt),
        ];
    }

    /**
     * Sidebar parts in the saved order, hidden ones included (for the admin form).
     *
     * @return list<array{key: string, label: string, note: ?string, visible: bool}>
     */
    public static function sidebarRows(): array
    {
        $s = Settings::all();
        $saved = self::csv($s['article_sidebar_order'] ?? '');
        $hidden = self::csv($s['article_sidebar_hidden'] ?? '');

        return array_map(
            fn (string $key) => [
                'key' => $key,
                'label' => self::WIDGETS[$key]['label'],
                'note' => self::WIDGETS[$key]['note'] ?? null,
                'visible' => in_array($key, $saved, true) && ! in_array($key, $hidden, true),
            ],
            self::orderedKeys($s['article_sidebar_order'] ?? '')
        );
    }

    /**
     * Turns the submitted sidebar rows into the two stored settings.
     *
     * @param  array<string, array<string, mixed>>  $rows  widgets[key][serial|show]
     * @return array{article_sidebar_order: string, article_sidebar_hidden: string}
     */
    public static function sidebarValues(array $rows): array
    {
        $sort = [];
        $position = 0;

        foreach (array_keys(self::WIDGETS) as $key) {
            $position++;
            $serial = (int) ($rows[$key]['serial'] ?? $position * 10);
            $sort[$key] = [$serial, $position];
        }

        uasort($sort, fn (array $a, array $b) => $a <=> $b);

        $order = array_keys($sort);
        $hidden = array_values(array_filter(
            $order,
            fn (string $key) => ! isset($rows[$key]['show'])
        ));

        return [
            'article_sidebar_order' => implode(',', $order),
            'article_sidebar_hidden' => implode(',', $hidden),
        ];
    }

    /** @param array<string, string> $opt */
    private static function css(array $opt): string
    {
        $vars = [
            '--sa-amber' => $opt['accent'],
            '--sa-link' => $opt['link'],
            '--sa-link-hover' => 'color-mix(in srgb, '.$opt['link'].' 75%, #000)',
            '--sa-font-display' => self::FONTS[$opt['font']]['css'],
            '--sa-title-size' => self::TITLE_SIZES[$opt['title_size']]['css'],
            '--sa-body-size' => self::BODY_SIZES[$opt['body_size']]['css'],
            '--sa-body-width' => self::BODY_WIDTHS[$opt['body_width']]['css'],
            '--sa-body-leading' => self::LEADINGS[$opt['leading']]['css'],
            '--sa-image-ratio' => self::IMAGE_RATIOS[$opt['image_ratio']]['css'],
            '--sa-image-fit' => self::IMAGE_FITS[$opt['image_fit']]['css'],
        ];

        $css = '';

        foreach ($vars as $name => $value) {
            $css .= $name.':'.$value.';';
        }

        return ':root{'.$css.'}';
    }

    /** @param array<string, mixed> $allowed */
    private static function pick(mixed $value, array $allowed, string $fallback): string
    {
        return is_string($value) && array_key_exists($value, $allowed) ? $value : $fallback;
    }

    /** @param array{min: int, max: int, default: int} $rule */
    private static function clamp(mixed $value, array $rule): int
    {
        $number = is_numeric($value) ? (int) $value : $rule['default'];

        return max($rule['min'], min($rule['max'], $number));
    }

    /** @return list<string> known widget keys, de-duplicated, in the given order */
    private static function csv(string $value): array
    {
        $keys = array_map('trim', explode(',', $value));

        return array_values(array_unique(array_filter(
            $keys,
            fn (string $key) => isset(self::WIDGETS[$key])
        )));
    }

    /** @return list<string> saved order first, then any widget missing from it */
    private static function orderedKeys(string $saved): array
    {
        $keys = self::csv($saved);

        return array_merge($keys, array_values(array_diff(array_keys(self::WIDGETS), $keys)));
    }
}

            const queueUrl = @json($queueUrl);
            const form = document.getElementById('pc-defaults');

            const text = document.getElementById('pc-progress-text');

                    logo: form.elements.card_logo.checked,
                    category: form.elements.card_category.checked,
                    date: form.elements.card_date.checked,

                    badge: '',
                    cta: '',
                    kicker: item.byline || '',
                    excerpt: true,
                    caption: true,
                    social: true
                };
            }

            function note(message) {
                log.textContent += message + '\n';
                log.scrollTop = log.scrollHeight;
            }

            btn.addEventListener('click', async function () {
                btn.disabled = true;
                log.textContent = '';
                bar.style.width = '0';
                text.textContent = 'তালিকা আনা হচ্ছে…';

                let data;

                try {
                    const res = await fetch(queueUrl, { credentials: 'same-origin', headers: { 'Accept': 'application/json' } });

                    if (!res.ok) { throw new Error('HTTP ' + res.status); }

                    data = await res.json();
                } catch (e) {
                    text.textContent = 'তালিকা আনা যায়নি: ' + e.message;
                    btn.disabled = false;
                    return;
                }

                if (!data.items.length) {
                    text.textContent = 'সব প্রকাশিত খবরের কার্ড আগেই তৈরি আছে।';
                    btn.disabled = false;
                    return;
                }

                const footer = form.elements.card_footer.value.trim() || data.footer;
                const canvas = document.createElement('canvas');
                let done = 0;
                let failed = 0;

                for (let i = 0; i < data.items.length; i++) {
                    const item = data.items[i];

                    text.textContent = (i + 1) + ' / ' + data.items.length + ' তৈরি হচ্ছে…';

                    try {
                        await PhotoCard.render(canvas, {
                            title: item.title,
                            category: item.category,
                            date: item.date,
                            image: item.image,
                            logo: data.logo,
                            emblem: data.emblem,
                            siteName: data.siteName,
                            tagline: data.tagline,
                            footer: footer,
                            excerpt: item.excerpt,
                            caption: item.caption,
                            byline: item.byline,
                            social: data.social
                        }, options(item));

                        const blob = await PhotoCard.toBlob(canvas, 'image/jpeg', 0.92);
                        const fd = new FormData();

                        fd.append('card', blob, 'card.jpg');

                        const res = await fetch(item.storeUrl, {
                            method: 'POST',
                            body: fd,
                            credentials: 'same-origin',
                            headers: { 'X-CSRF-TOKEN': token, 'Accept': 'application/json', 'X-Requested-With': 'XMLHttpRequest' }
                        });

                        if (!res.ok) { throw new Error('HTTP ' + res.status); }

                        done++;
                    } catch (e) {
                        failed++;
                        note('ব্যর্থ: #' + item.id + ' ' + item.title.slice(0, 40) + ' (' + e.message + ')');
                    }

                    bar.style.width = Math.round(((i + 1) / data.items.length) * 100) + '%';
                }

<?php

namespace App\Console\Commands;

use App\Models\Media;
use Illuminate\Console\Command;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\File;

class ImportMedia extends Command
{
    protected $signature = 'media:import';

    protected $description = 'uploads/news ফোল্ডারের বিদ্যমান ছবি গ্যালারিতে যোগ করে (বারবার চালানো নিরাপদ)';

    public function handle(): int
    {
        $dir = public_path(Media::DIR);

        if (! File::isDirectory($dir)) {
            $this->warn('ছবির ফোল্ডার পাওয়া যায়নি: '.$dir);

            return self::SUCCESS;
        }

        $known = Media::pluck('filename')->flip();
        $added = 0;

        foreach (File::files($dir) as $file) {
            $name = $file->getFilename();

            if ($known->has($name) || ! in_array(strtolower($file->getExtension()), ['jpg', 'jpeg', 'png', 'webp'], true)) {
                continue;
            }

            $time = Carbon::createFromTimestamp($file->getMTime());

            (new Media)->forceFill([
                'filename' => $name,
                'original_name' => $name,
                'mime' => File::mimeType($file->getPathname()) ?: null,
                'size' => $file->getSize(),
                'created_at' => $time,
                'updated_at' => $time,
            ])->save();

            $added++;
        }

        $this->info("{$added}টি ছবি গ্যালারিতে যোগ হয়েছে।");

        return self::SUCCESS;
    }
}

        $liveBySlot = Ad::where('status', true)->get()->filter->isLive()->groupBy('slot')->map->count();

    public function create(): View
    {
        return view('admin.ads.create', [

        ]);

    public function store(Request $request): RedirectResponse
    {
        $data = $this->payload($request);

        Ad::create($data + ['created_by' => $request->user()->id]);

        return redirect()->route('ads.index')->with('success', 'বিজ্ঞাপন যোগ হয়েছে।');
    }

    public function edit(Request $request, Ad $ad): View
    {
        $this->authorizeAd($request, $ad);

        return view('admin.ads.edit', compact('ad'));
    }

    {
        $this->authorizeAd($request, $ad);

        $oldImage = $ad->image;
        $ad->update($this->payload($request, $ad));

        if ($oldImage && $oldImage !== $ad->image) {
            $this->deleteImage($oldImage);
        }

        return redirect()->route('ads.index')->with('success', 'বিজ্ঞাপন আপডেট হয়েছে।');
    }

    public function destroy(Request $request, Ad $ad): RedirectResponse
    {
        $this->authorizeAd($request, $ad);

        $image = $ad->image;
        $ad->delete();
        $this->deleteImage($image);

        return redirect()->route('ads.index')->with('success', 'বিজ্ঞাপন মোছা হয়েছে।');

    /** কার্ডের নিচে ডানে যে লেখা বসবে: সেটিংয়ের লেখা, না থাকলে সাইটের ডোমেইন। */
    private function footerText(): string
    {
        $custom = Settings::get('card_footer');

        if ($custom !== '') {
            return $custom;
        }

        return (string) (parse_url((string) config('app.url'), PHP_URL_HOST) ?? '');
    }

    /** পূর্ণ ঠিকানা থেকে শুধু পথ (একই উৎসের ছবি, ক্যানভাস নিরাপদ থাকে)। */
    private function path(string $url): string
    {
        return parse_url($url, PHP_URL_PATH) ?: $url;

    /** HTML থেকে সাধারণ লেখা: ট্যাগ বাদ, সংক্ষিপ্ত করা। */
    private function plain(?string $html, int $limit): string
    {
        $text = html_entity_decode(strip_tags((string) $html), ENT_QUOTES | ENT_HTML5, 'UTF-8');

        return Str::limit(trim((string) preg_replace('/\s+/u', ' ', $text)), $limit, '…');
    }

    private function caption(News $news): string
    {
        $caption = trim((string) $news->image_caption);
        $credit = trim((string) $news->image_credit);

        return trim($caption.($credit !== '' ? ' (ছবি: '.$credit.')' : ''));
    }

    /** গোল প্রতীকের জন্য ফেভিকন (সেট করা থাকলে)। */
    private function emblem(): ?string
    {

        return $url ? $this->path($url) : null;
    }

    /**
     * সাইট সেটিংয়ের সোশ্যাল লিংক থেকে হ্যান্ডেলের নাম (যেমন facebook.com/abc হলে "abc")।
     *
     * @return list<array{type: string, text: string}>
     */
    private function social(): array
    {
        $items = [];

        foreach (['facebook' => 'social_facebook', 'instagram' => 'social_instagram', 'youtube' => 'social_youtube', 'x' => 'social_x'] as $type => $key) {
            $url = Settings::get($key);

            if ($url === '') {
                continue;
            }

            $path = trim((string) parse_url($url, PHP_URL_PATH), '/');
            $handle = ltrim(Str::afterLast($path, '/'), '@');

            $items[] = [
                'type' => $type,
                'text' => $handle !== '' ? $handle : (string) parse_url($url, PHP_URL_HOST),
            ];
        }

        return $items;
    }

    private function deleteFile(?string $name): void
    {
        if ($name) {
            File::delete(public_path(self::DIR.'/'.basename($name)));
        }
    }

    public function store(NewsRequest $request): RedirectResponse
    {
        $user = $request->user();
        $status = $request->validated('status');

        $data = $this->payload($request);
        $categoryIds = $request->validated('category_ids');

        // Keep the first selected category in the legacy column.
        $data['category_id'] = $categoryIds[0];

        $data['slug'] = News::uniqueSlug($data['title']);
        $data['author_id'] = $user->id;
        $data['reporter'] = ($data['reporter'] ?? null) ?: $user->name;
        $data['status'] = $status;
        $data['published_at'] = NewsWorkflow::resolvePublishedAt(
            $status,
            null,
            $request->publishAt()
        );
        $data['location_id'] = $request->location()?->id;

        if (! $user->isEditorOrAdmin()) {
            // Reporters cannot flag articles.
            $data['breaking'] = false;
            $data['featured'] = false;
            $data['is_lead'] = false;
        }

        $newImage = $request->hasFile('image')
            ? $this->storeImage($request->file('image'))
            : null;

        $data['image'] = $newImage ?: $this->galleryChoice($request);

        try {
            $news = DB::transaction(function () use (
                $data,
                $categoryIds
            ) {
                $news = News::create($data);

                $news->categories()->sync($categoryIds);

                return $news;
            });
        } catch (Throwable $e) {
            $this->deleteImage($newImage);

            throw $e;
        }

        if ($request->tagsSubmitted()) {
            $news->syncTagNames($request->validated('tags', []));
        }

        ArticleRevision::capture($news, $user, 'প্রথম সংস্করণ');

        Activity::record(
            'article.created',
            $news,
            ['status' => $status],
            $user
        );

            ->with(
                'success',

    public function update(
        NewsRequest $request,
        News $news
    ): RedirectResponse {
        $user = $request->user();

        // Keep the original image until the update succeeds.
        $oldImage = $news->image;

        // Do not regenerate the slug: existing links must keep working.
        $data = $this->payload($request);
        $categoryIds = $request->validated('category_ids');

        // Maintain the legacy category_id column.
        $data['category_id'] = $categoryIds[0];

        if (! $user->isEditorOrAdmin()) {
            // Preserve editorial flags set by an editor.
            unset(
                $data['breaking'],
                $data['featured'],
                $data['is_lead']
            );
        }

        $newImage = $request->hasFile('image')
            ? $this->storeImage($request->file('image'))
            : null;

        $picked = $newImage
            ? null
            : $this->galleryChoice($request);

        if ($newImage || $picked) {
            $data['image'] = $newImage ?: $picked;
        }

        // Keep the existing location unless the form submitted this field.
        if ($request->locationSubmitted()) {
            $data['location_id'] = $request->location()?->id;
        }

        $target = $request->validated('status');
        $changesStatus = $target && $target !== $news->status;

        try {
            DB::transaction(function () use (
                $news,
                $data,
                $categoryIds,
                $request,
                $user,
                $target,
                $changesStatus
            ) {
                $news->update($data);

                $news->categories()->sync($categoryIds);

                if ($request->tagsSubmitted()) {
                    $news->syncTagNames(
                        $request->validated('tags', [])
                    );
                }

                if ($changesStatus) {
                    NewsWorkflow::transition(
                        $news,
                        $user,
                        $target,
                        null,
                        $request->publishAt()
                    );
                } else {
                    ArticleRevision::capture($news, $user);
                }
            });
        } catch (Throwable $e) {
            // Only the newly uploaded image is cleaned up on failure.
            $this->deleteImage($newImage);

            throw $e;
        }

        // Remove the old image only after a successful replacement.
        // Never delete it if another article still uses it.
        if ($newImage && $oldImage && $oldImage !== $newImage) {
            $this->deleteImageIfUnused($oldImage, $news);
        }

        Activity::record('article.updated', $news, [], $user);

        return redirect()
            ->route('news.index')
            ->with('success', 'সংবাদ আপডেট হয়েছে।');
    }

    /**
     * Remove an image uploaded during an operation that failed.
     *
     * This helper is used for newly uploaded files that have not
     * been successfully attached to an article.
     */
    private function deleteImage(?string $name): void
    {
        if (! $name) {
            return;
        }

        $name = basename($name);

        $path = public_path(self::IMAGE_DIR.'/'.$name);

        if (File::exists($path)) {
            File::delete($path);
        }

        Media::where('filename', $name)->delete();
    }

    /**
     * Delete a replaced image only when no other article references it.
     */
    private function deleteImageIfUnused(
        string $name,
        News $updatedNews
    ): void {
        $name = basename($name);

        $stillUsed = News::withTrashed()
            ->where('id', '!=', $updatedNews->id)
            ->where(function ($query) use ($name) {
                $query->where('image', $name)
                    ->orWhere('card_image', $name);
            })
            ->exists();

        if ($stillUsed) {
            return;
        }

        $path = public_path(self::IMAGE_DIR.'/'.$name);

        if (File::exists($path)) {
            File::delete($path);
        }

        Media::where('filename', $name)->delete();
    }

    private function galleryChoice(Request $request): ?string
    {
        $name = trim(
            (string) $request->input('gallery_image', '')
        );

        return $name !== ''
            ? Media::where('filename', $name)->value('filename')
            : null;
    }

            'ticker_count.max' => 'টিকারে সর্বোচ্চ ২০টি খবর রাখা যাবে।',
        ]);

        $current = Settings::all();
        $values = [];

        foreach ([
            'site_name', 'site_tagline', 'footer_about', 'footer_email', 'footer_phone', 'footer_address',
            'social_facebook', 'social_youtube', 'social_x', 'social_instagram',
            'footer_copyright', 'footer_credit', 'ticker_label',
        ] as $key) {
            $values[$key] = trim((string) $request->input($key, ''));
        }

        foreach (['footer_categories', 'ticker_count', 'ticker_max_hours'] as $key) {
            $values[$key] = (string) (int) $request->input($key);
        }

        $values['ticker_speed'] = (string) $request->input('ticker_speed');
        $values['show_date'] = $request->boolean('show_date') ? '1' : '0';
        $values['ticker_show'] = $request->boolean('ticker_show') ? '1' : '0';
        $values['footer_links'] = $this->cleanLinks((string) $request->input('footer_links', ''));

        $values['logo'] = $this->image($request, 'logo', 'remove_logo', $current['logo']);
        $values['favicon'] = $this->image($request, 'favicon', 'remove_favicon', $current['favicon']);

        Settings::set($values);

        return back()->with('success', 'সাইটের সেটিং সংরক্ষণ হয়েছে।');
    }

    /** প্রতি লাইনে "লেবেল | লিংক"; ভুল লাইন থাকলে সেই লাইনের নম্বর বলে দেয়। */
    private function cleanLinks(string $raw): string
    {
        $out = [];

        foreach (preg_split('/\r\n|\r|\n/', $raw) as $index => $line) {
            $line = trim($line);

            if ($line === '') {
                continue;
            }

            $parts = array_map('trim', explode('|', $line, 2));

            if (count($parts) !== 2 || $parts[0] === '' || ! preg_match('~^(https?://|/(?!/))\S+$~u', $parts[1])) {
                throw ValidationException::withMessages([
                    'footer_links' => 'ফুটারের লিংকের '.($index + 1).' নম্বর লাইন ঠিক নয়। এভাবে লিখুন: লেবেল | https://example.com অথবা লেবেল | /about',
                ]);
            }

            $out[] = $parts[0].'|'.$parts[1];
        }

        return implode("\n", array_slice($out, 0, 12));
    }

    /** নতুন ছবি দিলে সেটি রাখে ও পুরোনোটি মোছে; "মুছুন" টিক দিলে ফাইল মুছে ডিফল্টে ফেরে। */
    private function image(Request $request, string $field, string $removeField, string $old): string
    {
        if ($request->hasFile($field)) {
            $file = $request->file($field);
            $name = $field.'-'.Str::uuid().'.'.$file->extension();
            $file->move(public_path(Settings::DIR), $name);

            $this->deleteFile($old);

            return $name;
        }

        if ($request->boolean($removeField)) {
            $this->deleteFile($old);

            return '';
        }

        return $old;
    }

    private function deleteFile(string $name): void
    {
        if ($name !== '') {
            File::delete(public_path(Settings::DIR.'/'.basename($name)));
        }
    }
}

    {
        return ['size' => 'integer'];
    }

    public function uploader(): BelongsTo
    {
        return $this->belongsTo(User::class, 'uploaded_by');
    }

    public function getUrlAttribute(): string
    {
        return asset(self::DIR.'/'.$this->filename);
    }

    public function getSizeLabelAttribute(): string
    {
        return $this->size >= 1048576
            ? number_format($this->size / 1048576, 1).' MB'
            : number_format($this->size / 1024).' KB';
    }

    /**
     * ফাইল সংরক্ষণ করে গ্যালারিতে যোগ করে। নাম এলোমেলো (UUID), ক্লায়েন্টের ফাইলনাম কখনো ব্যবহার হয় না।
     */
    public static function storeUpload(UploadedFile $file, ?User $user = null): self
    {
        // move() এর পরে ফাইলের তথ্য পাওয়া যায় না, তাই আগেই নিয়ে রাখছি।
        $name = Str::uuid().'.'.$file->extension();
        $original = Str::limit($file->getClientOriginalName(), 190, '');
        $mime = $file->getMimeType();
        $size = (int) $file->getSize();

        $file->move(public_path(self::DIR), $name);

        return static::create([
            'filename' => $name,
            'original_name' => $original,
            'mime' => $mime,
            'size' => $size,
            'uploaded_by' => $user?->id,
        ]);
    }

<?php

namespace App\Support;

use App\Models\News;
use App\Models\User;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Str;

final class Avatars
{
    public const DIR = 'uploads/avatars';

    public static function url(?User $user): ?string
    {
        return ($user && $user->avatar) ? asset(self::DIR.'/'.$user->avatar) : null;
    }

    /** নতুন ছবি রেখে আগেরটি মোছে। */
    public static function save(User $user, UploadedFile $file): void
    {
        $name = $user->id.'-'.Str::uuid().'.'.$file->extension();
        $file->move(public_path(self::DIR), $name);

        $old = $user->avatar;

        // forceFill: ইউজার মডেলের fillable তালিকা না ছুঁয়ে কলামটি সংরক্ষণ
        $user->forceFill(['avatar' => $name])->save();

        self::delete($old);
    }

    public static function remove(User $user): void
    {
        $old = $user->avatar;

        $user->forceFill(['avatar' => null])->save();

        self::delete($old);
    }

    /**
     * খবরের পেজের জন্য লেখকের নাম, ছবি ও পরিচিতি।
     * বাইলাইনের নাম লেখক-ইউজারের নামের সাথে না মিললে (অতিথি লেখক) ছবি ও পরিচিতি দেওয়া হয় না।
     *
     * @return array{name: string, photo: ?string, bio: string}
     */
    public static function byline(News $news): array
    {
        $author = $news->author;
        $name = trim((string) $news->reporter) ?: trim((string) ($author->name ?? ''));
        $same = $author && mb_strtolower($name) === mb_strtolower(trim((string) $author->name));

        return [
            'name' => $name,
            'photo' => $same ? self::url($author) : null,
            'bio' => $same ? trim((string) ($author->bio ?? '')) : '',
        ];
    }

    private static function delete(?string $name): void
    {
        if ($name) {
            File::delete(public_path(self::DIR.'/'.basename($name)));
        }
    }
}
